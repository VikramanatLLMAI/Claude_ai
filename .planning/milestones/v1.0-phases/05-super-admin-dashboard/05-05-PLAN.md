**2. Service (lib/services/platform-settings-service.ts):**

Uses raw `prisma` (platform-level, not tenant-scoped).

- `getPlatformSettings()`: Upsert pattern -- find or create singleton record. Return PlatformSettings.
- `updatePlatformSettings(data, userId, ipAddress)`: Update the singleton. Audit log the change with before/after diff in metadata. Return updated record.

```typescript
import prisma from "@/lib/db"
import { auditLog } from "@/lib/services/audit-service"

export async function getPlatformSettings() {
  return prisma.platformSettings.upsert({
    where: { id: "singleton" },
    update: {},
    create: { id: "singleton" },
  })
}

export async function updatePlatformSettings(
  data: Partial<PlatformSettingsInput>,
  userId: string,
  ipAddress: string | null
) {
  return prisma.$transaction(async (tx) => {
    const current = await tx.platformSettings.findUnique({ where: { id: "singleton" } })
    const updated = await tx.platformSettings.upsert({
      where: { id: "singleton" },
      update: data,
      create: { id: "singleton", ...data },
    })
    await auditLog.record(tx, {
      action: "platform_settings.updated",
      userId,
      metadata: { before: current, after: updated },
      ipAddress,
    })
    return updated
  })
}
```
