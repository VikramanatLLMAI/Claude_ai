   function parseToolSource(name: string): { source: 'org' | 'role' | null; cleanName: string } {
     const match = name.match(/^mcp__(org|role)__(.+)$/);
     if (match) {
       return { source: match[1] as 'org' | 'role', cleanName: match[2] };
     }
     return { source: null, cleanName: name };
   }
