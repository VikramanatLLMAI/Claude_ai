  id               String   @id @default(uuid())

  loginBadge       String?  @map("login_badge")
  loginDescription String?  @map("login_description") @db.Text
  loginFeatureCards Json    @default("[]") @map("login_feature_cards")
  createdAt        DateTime @default(now()) @map("created_at")
  updatedAt        DateTime @updatedAt @map("updated_at")

  organization Organization @relation(fields: [organizationId], references: [id], onDelete: Cascade)

  @@map("login_brandings")
}

   - `export function getIcon(name: string): LucideIcon` - returns icon component or Sparkles fallback
   - `export function getIconNames(): string[]` - returns sorted list of available icon names
