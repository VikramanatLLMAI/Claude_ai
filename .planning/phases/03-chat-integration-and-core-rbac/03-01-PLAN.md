     ```typescript
     export const TOKEN_LIMITS = { org: 700, role: 500, user: 200 } as const;
     export const SERVER_MARGIN = 1.05; // Accept up to 105% to account for approximation error
     export function estimateTokenCount(text: string): number {
       if (!text || text.trim().length === 0) return 0;
       return Math.ceil(text.length / 4); // ~4 English chars per token
     }
     ```

  6. Create `lib/prompt-sanitizer.ts` -- dedicated sanitization utility for untrusted prompt inputs (PRMT-06). This is created here in Plan 01 so it is available to all Wave 2 plans (Plans 03, 04, 05):
     ```typescript
     /**
      * Sanitize untrusted prompt input for safe injection into XML-delimited system prompt.
      * Two-step approach:
      * 1. Strip all XML-like tags (prevents delimiter breaking)
      * 2. Escape <, >, & characters (prevents injection)
      */
     export function sanitizePromptLayer(text: string): string {
       return text
         .replace(/<[^>]*>/g, '')        // Strip XML tags
         .replace(/&/g, '&amp;')         // Escape ampersand first
         .replace(/</g, '&lt;')          // Escape less-than
         .replace(/>/g, '&gt;');         // Escape greater-than
     }
     ```
